#include <afxwin.h>
#include <atlimage.h>
#include <math.h>
#include <tchar.h>

const double Pi = 3.14159265358979323846;

int RotateX(double x, double y, CPoint ptCenter, float angle)
{
    double angleRad = angle * Pi / 180.0;
    double cosAngle = cos(angleRad);
    double sinAngle = sin(angleRad);

    return (int)(ptCenter.x + (x - ptCenter.x) * cosAngle - (y - ptCenter.y) * sinAngle);
}

int RotateY(double x, double y, CPoint ptCenter, float angle)
{
    double angleRad = angle * Pi / 180.0;
    double cosAngle = cos(angleRad);
    double sinAngle = sin(angleRad);

    return (int)(ptCenter.y + (x - ptCenter.x) * sinAngle + (y - ptCenter.y) * cosAngle);
}

void DrawRotatedLineTo(
    CDC* pDC,
    double x,
    double y,
    CPoint ptCenter,
    float lakeRotation)
{
    pDC->LineTo(
        RotateX(x, y, ptCenter, lakeRotation),
        RotateY(x, y, ptCenter, lakeRotation));
}

void DrawArcPart(
    CDC* pDC,
    double& currentX,
    double& currentY,
    double& direction,
    int radius,
    double angle,
    CPoint ptRotationCenter,
    float lakeRotation)
{
    double centerX = currentX - radius * sin(direction);
    double centerY = currentY + radius * cos(direction);

    double startX = currentX;
    double startY = currentY;

    int segmentCount = (int)(angle / (Pi / 36.0));

    if (segmentCount < 8)
        segmentCount = 8;

    for (int i = 1; i <= segmentCount; i++)
    {
        double currentAngle = angle * i / segmentCount;
        double cosAngle = cos(currentAngle);
        double sinAngle = sin(currentAngle);

        double x = startX - centerX;
        double y = startY - centerY;

        double nextX = centerX + x * cosAngle - y * sinAngle;
        double nextY = centerY + x * sinAngle + y * cosAngle;

        DrawRotatedLineTo(
            pDC,
            nextX,
            nextY,
            ptRotationCenter,
            lakeRotation);
    }

    double cosEndAngle = cos(angle);
    double sinEndAngle = sin(angle);

    double endX = startX - centerX;
    double endY = startY - centerY;

    currentX = centerX + endX * cosEndAngle - endY * sinEndAngle;
    currentY = centerY + endX * sinEndAngle + endY * cosEndAngle;

    direction += angle;
}

void DrawLakePart(
    CDC* pDC,
    wchar_t part,
    double& currentX,
    double& currentY,
    double& direction,
    int radius,
    CPoint ptRotationCenter,
    float lakeRotation)
{
    switch (part)
    {
        case L'1':
            DrawArcPart(
                pDC,
                currentX,
                currentY,
                direction,
                radius,
                Pi / 2.0,
                ptRotationCenter,
                lakeRotation);
            break;

        case L'2':
            DrawArcPart(
                pDC,
                currentX,
                currentY,
                direction,
                radius,
                Pi,
                ptRotationCenter,
                lakeRotation);
            break;

        case L'3':
            DrawArcPart(
                pDC,
                currentX,
                currentY,
                direction,
                radius,
                3.0 * Pi / 2.0,
                ptRotationCenter,
                lakeRotation);
            break;

        case L'4':
            DrawArcPart(
                pDC,
                currentX,
                currentY,
                direction,
                radius,
                2.0 * Pi,
                ptRotationCenter,
                lakeRotation);
            break;

        case L'5':
            direction = 0.0;
            currentX += radius;

            DrawRotatedLineTo(
                pDC,
                currentX,
                currentY,
                ptRotationCenter,
                lakeRotation);
            break;

        case L'6':
            direction = Pi / 2.0;
            currentY += radius;

            DrawRotatedLineTo(
                pDC,
                currentX,
                currentY,
                ptRotationCenter,
                lakeRotation);
            break;
    }
}

void DrawLakeInternal(
    CDC* pDC,
    CString strParts,
    CPoint ptStart,
    int radius,
    COLORREF colFill,
    COLORREF colLine,
    CString strText,
    CPoint ptTextCenter,
    COLORREF colText,
    float lakeRotation,
    float textRotation)
{
    if (pDC == NULL || radius <= 0 || strParts.IsEmpty())
        return;

    CPen pen(PS_SOLID, 1, colLine);
    CBrush brush(colFill);

    CPen* oldPen = pDC->SelectObject(&pen);
    CBrush* oldBrush = pDC->SelectObject(&brush);

    double currentX = ptStart.x;
    double currentY = ptStart.y;
    double direction = 0.0;

    pDC->BeginPath();

    pDC->MoveTo(
        RotateX(currentX, currentY, ptStart, lakeRotation),
        RotateY(currentX, currentY, ptStart, lakeRotation));

    for (int i = 0; i < strParts.GetLength(); i++)
    {
        DrawLakePart(
            pDC,
            strParts[i],
            currentX,
            currentY,
            direction,
            radius,
            ptStart,
            lakeRotation);
    }

    pDC->CloseFigure();
    pDC->EndPath();
    pDC->StrokeAndFillPath();

    pDC->SelectObject(oldBrush);
    pDC->SelectObject(oldPen);

    LOGFONT logFont = {};
    logFont.lfHeight = -MulDiv(10, pDC->GetDeviceCaps(LOGPIXELSY), 72);
    logFont.lfWeight = FW_BOLD;
    logFont.lfEscapement = (LONG)(textRotation * 10);
    logFont.lfOrientation = logFont.lfEscapement;

    _tcscpy_s(logFont.lfFaceName, LF_FACESIZE, _T("Arial"));

    CFont font;
    font.CreateFontIndirect(&logFont);

    CFont* oldFont = pDC->SelectObject(&font);
    int oldBkMode = pDC->SetBkMode(TRANSPARENT);
    COLORREF oldTextColor = pDC->SetTextColor(colText);

    CPoint ptRotatedTextCenter(
        RotateX(ptTextCenter.x, ptTextCenter.y, ptStart, lakeRotation),
        RotateY(ptTextCenter.x, ptTextCenter.y, ptStart, lakeRotation));

    CSize textSize = pDC->GetTextExtent(strText);

    pDC->TextOut(
        ptRotatedTextCenter.x - textSize.cx / 2,
        ptRotatedTextCenter.y - textSize.cy / 2,
        strText);

    pDC->SetTextColor(oldTextColor);
    pDC->SetBkMode(oldBkMode);
    pDC->SelectObject(oldFont);
}

void DrawLake(
    CDC* pDC,
    CString strParts,
    CPoint ptStart,
    int radius,
    COLORREF colFill,
    COLORREF colLine,
    CString strText,
    CPoint ptTextCenter,
    COLORREF colText,
    float angleRotation)
{
    DrawLakeInternal(
        pDC,
        strParts,
        ptStart,
        radius,
        colFill,
        colLine,
        strText,
        ptTextCenter,
        colText,
        0,
        angleRotation);
}

void DrawLakes(
    CDC* pDC,
    CString arParts[],
    CPoint arPtStart[],
    int arRadius[],
    COLORREF arColFill[],
    COLORREF arColLine[],
    CString arStrText[],
    CPoint arPtTextCenter[],
    COLORREF arColText[],
    float arRotation[],
    int length)
{
    if (pDC == NULL)
        return;

    for (int i = 0; i < length; i++)
    {
        DrawLakeInternal(
            pDC,
            arParts[i],
            arPtStart[i],
            arRadius[i],
            arColFill[i],
            arColLine[i],
            arStrText[i],
            arPtTextCenter[i],
            arColText[i],
            arRotation[i],
            0);
    }
}

void SetPicture(CDC* pDC, CRect rcView, CString strPicture)
{
    if (pDC == NULL || rcView.Width() <= 0 || rcView.Height() <= 0)
        return;

    CImage image;

    image.Create(
        rcView.Width(),
        rcView.Height(),
        32);

    BitBlt(
        image.GetDC(),
        0,
        0,
        rcView.Width(),
        rcView.Height(),
        pDC->GetSafeHdc(),
        rcView.left,
        rcView.top,
        SRCCOPY);

    image.ReleaseDC();
    image.Save(strPicture);
}
